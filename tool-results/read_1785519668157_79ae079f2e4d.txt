   214→   214→function calculateStaticPremium(input: {
   215→   215→  brand: string;
   216→   216→  modelDescription: string;
   217→   217→  vehicleYear: number;
   218→   218→  coverageType: "Comprehensive" | "TLO";
   219→   219→  plateCode: string;
   220→   220→  addOns: string[];
   221→   221→  vehicleValueOverride?: number;
   222→   222→  vehicleTypeCategory?: string;
   223→   223→  tplCoverageAmount?: number;
   224→   224→  paDriverAmount?: number;
   225→   225→  paPassengerCount?: number;
   226→   226→}) {
   227→   227→  const CURRENT_YEAR = new Date().getFullYear();
   228→   228→
   229→   229→  // 1. Get vehicle value from static JSON
   230→   230→  let vehicleValue = input.vehicleValueOverride || 0;
   231→   231→  if (!vehicleValue) {
   232→   232→    const brandData = vehiclePriceData[input.brand as keyof typeof vehiclePriceData];
   233→   233→    if (brandData) {
   234→   234→      const modelKey = Object.keys(brandData).find(
   235→   235→        (m) => m.toLowerCase().includes(input.modelDescription.toLowerCase()) ||
   236→   236→               input.modelDescription.toLowerCase().includes(m.toLowerCase())
   237→   237→      );
   238→   238→      if (modelKey) {
   239→   239→        const yearData = brandData[modelKey as keyof typeof brandData];
   240→   240→        const price = yearData[String(input.vehicleYear) as keyof typeof yearData];
   241→   241→        if (typeof price === "number") {
   242→   242→          vehicleValue = price * 1_000_000; // Convert from millions
   243→   243→        }
   244→   244→      }
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  if (!vehicleValue) {
   249→   249→    return {
   250→   250→      success: false as const,
   251→   251→      error: "Harga kendaraan tidak ditemukan. Silakan masukkan harga OTR manual.",
   252→   252→    };
   253→   253→  }
   254→   254→
   255→   255→  // 2. Vehicle age
   256→   256→  const vehicleAge = CURRENT_YEAR - input.vehicleYear;
   257→   257→
   258→   258→  // 3. Wilayah from static plate map
   259→   259→  const plateUpper = input.plateCode.toUpperCase().trim();
   260→   260→  const plateInfo = STATIC_PLATE_MAP[plateUpper] || { city: "Jakarta", wilayah: 2 };
   261→   261→  const wilayah = plateInfo.wilayah;
   262→   262→  const plateCity = plateInfo.city;
   263→   263→
   264→   264→  // 4. Vehicle type category
   265→   265→  const vehicleTypeCategory = input.vehicleTypeCategory || "Non Bus dan Non Truk";
   266→   266→
   267→   267→  // 5. Category
   268→   268→  let category: number;
   269→   269→  if (vehicleTypeCategory === "Non Bus dan Non Truk") {
   270→   270→    if (vehicleValue <= 125_000_000) category = 1;
   271→   271→    else if (vehicleValue <= 200_000_000) category = 2;
   272→   272→    else if (vehicleValue <= 400_000_000) category = 3;
   273→   273→    else if (vehicleValue <= 800_000_000) category = 4;
   274→   274→    else category = 5;
   275→   275→  } else if (vehicleTypeCategory === "Truk dan Pick Up") {
   276→   276→    category = 6;
   277→   277→  } else if (vehicleTypeCategory === "Bus") {
   278→   278→    category = 7;
   279→   279→  } else {
   280→   280→    category = 8;
   281→   281→  }
   282→   282→
   283→   283→  // 6. Get rate
   284→   284→  const catKey = `cat${category}`;
   285→   285→  const rateTable = input.coverageType === "TLO" ? STATIC_TLO_RATES : STATIC_RATES;
   286→   286→  const typeRates = rateTable[vehicleTypeCategory] || rateTable["Non Bus dan Non Truk"];
   287→   287→  const rates = typeRates[catKey] || typeRates["cat3"] || [0.0293, 0.0237, 0.0190];
   288→   288→  const baseRate = wilayah === 1 ? rates[0] : wilayah === 2 ? rates[1] : rates[2];
   289→   289→
   290→   290→  // 7. Loading (for Comprehensive only, age > 5)
   291→   291→  let loadingRate = 0;
   292→   292→  if (input.coverageType === "Comprehensive" && vehicleAge > 5) {
   293→   293→    const loadingPercent = 0.05; // 5% per year above threshold
   294→   294→    loadingRate = baseRate * ((vehicleAge - 5) * loadingPercent);
   295→   295→  }
   296→   296→
   297→   297→  const effectiveRate = baseRate + loadingRate;
   298→   298→  const basePremium = Math.round(vehicleValue * effectiveRate);
   299→   299→
   300→   300→  // 8. Add-ons
   301→   301→  const addonPremiums: Array<{ key: string; label: string; rate: number; premium: number; coverageAmount: number }> = [];
   302→   302→
   303→   303→  for (const addonKey of input.addOns) {
   304→   304→    if (addonKey === "tpl") {
   305→   305→      // TPL for motorcycle = skip
   306→   306→      if (category === 8) continue;
   307→   307→      const tplCoverageAmount = input.tplCoverageAmount || 10_000_000;
   308→   308→      // Simplified TPL: ~1% of coverage
   309→   309→      const tplPremium = Math.round(tplCoverageAmount * 0.01);
   310→   310→      addonPremiums.push({
   311→   311→        key: "tpl",
   312→   312→        label: "Tanggung Jawab Pihak Ketiga (TPL)",
   313→   313→        rate: 0.01,
   314→   314→        premium: tplPremium,
   315→   315→        coverageAmount: tplCoverageAmount,
   316→   316→      });
   317→   317→      continue;
   318→   318→    }
   319→   319→
   320→   320→    if (addonKey === "paDriver") {
   321→   321→      const paDriverAmount = input.paDriverAmount || 10_000_000;
   322→   322→      const paDriverPremium = Math.round(paDriverAmount * 0.005);
   323→   323→      addonPremiums.push({
   324→   324→        key: "paDriver",
   325→   325→        label: "Perlindungan Jiwa Supir",
   326→   326→        rate: 0.005,
   327→   327→        premium: paDriverPremium,
   328→   328→        coverageAmount: paDriverAmount,
   329→   329→      });
   330→   330→      continue;
   331→   331→    }
   332→   332→
   333→   333→    if (addonKey === "paPassenger") {
   334→   334→      const paPassengerPremium = 40000;
   335→   335→      const paPassengerCount = input.paPassengerCount || 4;
   336→   336→      addonPremiums.push({
   337→   337→        key: "paPassenger",
   338→   338→        label: `Perlindungan Jiwa Penumpang (${paPassengerCount} orang)`,
   339→   339→        rate: 0,
   340→   340→        premium: paPassengerPremium,
   341→   341→        coverageAmount: paPassengerPremium,
   342→   342→      });
   343→   343→      continue;
   344→   344→    }
   345→   345→
   346→   346→    const staticAddon = STATIC_ADDON_RATES[addonKey];
   347→   347→    if (staticAddon) {
   348→   348→      const applies = staticAddon.appliesTo.includes(input.coverageType) || staticAddon.appliesTo.includes("All");
   349→   349→      if (applies) {
   350→   350→        addonPremiums.push({
   351→   351→          key: addonKey,
   352→   352→          label: staticAddon.label,
   353→   353→          rate: staticAddon.rate,
   354→   354→          premium: Math.round(vehicleValue * staticAddon.rate),
   355→   355→          coverageAmount: vehicleValue,
   356→   356→        });
   357→   357→      }
   358→   358→    }
   359→   359→  }
   360→   360→
   361→   361→  // 9. Totals
   362→   362→  const totalAddonPremium = addonPremiums.reduce((sum, a) => sum + a.premium, 0);
   363→   363→  const totalPremiumBeforeDiscount = basePremium + totalAddonPremium;
   364→   364→  const discountPercent = 25; // Default discount
   365→   365→  const discountAmount = Math.round(totalPremiumBeforeDiscount * (discountPercent / 100));
   366→   366→  const adminFee = 50000;
   367→   367→  const policyFee = 0;
   368→   368→  const totalPremium = totalPremiumBeforeDiscount - discountAmount + adminFee + policyFee;
   369→   369→
   370→   370→  // 10. OTR range
   371→   371→  const OTR_RANGE_PERCENT = 0.15;
   372→   372→  const otrMin = Math.round(vehicleValue * (1 - OTR_RANGE_PERCENT));
   373→   373→  const otrMax = Math.round(vehicleValue * (1 + OTR_RANGE_PERCENT));
   374→   374→  const formatRupiah = (amt: number) =>
   375→   375→    new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", minimumFractionDigits: 0, maximumFractionDigits: 0 }).format(amt);
   376→   376→
   377→   377→  // 11. Eligibility
   378→   378→  let isEligible = true;
   379→   379→  let ineligibilityReason: string | undefined;
   380→   380→  if (input.coverageType === "Comprehensive" && vehicleAge > 12) {
   381→   381→    isEligible = false;
   382→   382→    ineligibilityReason = `Kendaraan berusia ${vehicleAge} tahun. All Risk/Comprehensive maksimal 12 tahun.`;
   383→   383→  } else if (input.coverageType === "TLO" && vehicleAge > 15) {
   384→   384→    isEligible = false;
   385→   385→    ineligibilityReason = `Kendaraan berusia ${vehicleAge} tahun. TLO maksimal 15 tahun.`;
   386→   386→  }
   387→   387→
   388→   388→  // 12. Partner calculation
   389→   389→  const defaultModifiers: Record<string, number> = {
