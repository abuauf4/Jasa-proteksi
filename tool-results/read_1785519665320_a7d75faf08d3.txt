    85→// Fetch active partners from the database (with addon rate overrides)
    86→async function getActivePartnersFromDB() {
    87→  try {
    88→    const dbPartners = await db.insurancePartner.findMany({
    89→      where: { status: "active" },
    90→      orderBy: { sortOrder: "asc" },
    91→      include: { addonRateOverrides: { where: { isActive: true } } },
    92→    });
    93→    return dbPartners.map((p) => ({
    94→      name: p.name,
    95→      key: `partner${p.slug.replace(/-/g, "").replace(/\s+/g, "")}Modifier`,
    96→      modifier: p.modifier,
    97→      addonModifier: p.addonModifier ?? 1.0,
    98→      adminFee: p.adminFee ?? 50000,
    99→      bengkelResmiMaxYears: p.bengkelResmiMaxYears ?? null,
   100→      benefits: p.benefits ? JSON.parse(p.benefits) : [],
   101→      facilities: p.facilities ? JSON.parse(p.facilities) : [],
   102→      availableAddOns: DEFAULT_AVAILABLE_ADDONS,
   103→      addonRateOverrides: (p.addonRateOverrides || []).map(o => ({
   104→        addonKey: o.addonKey,
   105→        rate: o.rate,
   106→        addonLabel: o.addonLabel,
   107→      })),
   108→    }));
   109→  } catch {
   110→    return null;
   111→  }

   213→// ─── Static Fallback Calculation ───
   214→function calculateStaticPremium(input: {
   215→  brand: string;
   216→  modelDescription: string;
   217→  vehicleYear: number;
   218→  coverageType: "Comprehensive" | "TLO";
   219→  plateCode: string;
   220→  addOns: string[];
   221→  vehicleValueOverride?: number;
   222→  vehicleTypeCategory?: string;
   223→  tplCoverageAmount?: number;
   224→  paDriverAmount?: number;
   225→  paPassengerCount?: number;
   226→}) {
   227→  const CURRENT_YEAR = new Date().getFullYear();
   228→
   229→  // 1. Get vehicle value from static JSON
   230→  let vehicleValue = input.vehicleValueOverride || 0;
   231→  if (!vehicleValue) {
   232→    const brandData = vehiclePriceData[input.brand as keyof typeof vehiclePriceData];
   233→    if (brandData) {
   234→      const modelKey = Object.keys(brandData).find(
   235→        (m) => m.toLowerCase().includes(input.modelDescription.toLowerCase()) ||
   236→               input.modelDescription.toLowerCase().includes(m.toLowerCase())
   237→      );
   238→      if (modelKey) {
   239→        const yearData = brandData[modelKey as keyof typeof brandData];
   240→        const price = yearData[String(input.vehicleYear) as keyof typeof yearData];
   241→        if (typeof price === "number") {
   242→          vehicleValue = price * 1_000_000; // Convert from millions
   243→        }
   244→      }
   245→    }
   246→  }
   247→
   248→  if (!vehicleValue) {
   249→    return {
   250→      success: false as const,
   251→      error: "Harga kendaraan tidak ditemukan. Silakan masukkan harga OTR manual.",
   252→    };
   253→  }
   254→
   255→  // 2. Vehicle age
   256→  const vehicleAge = CURRENT_YEAR - input.vehicleYear;
   257→
   258→  // 3. Wilayah from static plate map
   259→  const plateUpper = input.plateCode.toUpperCase().trim();
   260→  const plateInfo = STATIC_PLATE_MAP[plateUpper] || { city: "Jakarta", wilayah: 2 };
   261→  const wilayah = plateInfo.wilayah;
   262→  const plateCity = plateInfo.city;
   263→
   264→  // 4. Vehicle type category
   265→  const vehicleTypeCategory = input.vehicleTypeCategory || "Non Bus dan Non Truk";
   266→
   267→  // 5. Category
   268→  let category: number;
   269→  if (vehicleTypeCategory === "Non Bus dan Non Truk") {
   270→    if (vehicleValue <= 125_000_000) category = 1;
   271→    else if (vehicleValue <= 200_000_000) category = 2;
   272→    else if (vehicleValue <= 400_000_000) category = 3;
   273→    else if (vehicleValue <= 800_000_000) category = 4;
   274→    else category = 5;
   275→  } else if (vehicleTypeCategory === "Truk dan Pick Up") {
   276→    category = 6;
   277→  } else if (vehicleTypeCategory === "Bus") {
   278→    category = 7;
   279→  } else {
   280→    category = 8;
   281→  }
   282→
   283→  // 6. Get rate
   284→  const catKey = `cat${category}`;
   285→  const rateTable = input.coverageType === "TLO" ? STATIC_TLO_RATES : STATIC_RATES;
   286→  const typeRates = rateTable[vehicleTypeCategory] || rateTable["Non Bus dan Non Truk"];
   287→  const rates = typeRates[catKey] || typeRates["cat3"] || [0.0293, 0.0237, 0.0190];
   288→  const baseRate = wilayah === 1 ? rates[0] : wilayah === 2 ? rates[1] : rates[2];
   289→
   290→  // 7. Loading (for Comprehensive only, age > 5)
   291→  let loadingRate = 0;
   292→  if (input.coverageType === "Comprehensive" && vehicleAge > 5) {
   293→    const loadingPercent = 0.05; // 5% per year above threshold
   294→    loadingRate = baseRate * ((vehicleAge - 5) * loadingPercent);
   295→  }
   296→
   297→  const effectiveRate = baseRate + loadingRate;
   298→  const basePremium = Math.round(vehicleValue * effectiveRate);
   299→
   300→  // 8. Add-ons
   301→  const addonPremiums: Array<{ key: string; label: string; rate: number; premium: number; coverageAmount: number }> = [];
   302→
   303→  for (const addonKey of input.addOns) {
   304→    if (addonKey === "tpl") {
   305→      // TPL for motorcycle = skip
   306→      if (category === 8) continue;
   307→      const tplCoverageAmount = input.tplCoverageAmount || 10_000_000;
   308→      // Simplified TPL: ~1% of coverage
   309→      const tplPremium = Math.round(tplCoverageAmount * 0.01);
   310→      addonPremiums.push({
   311→        key: "tpl",
   312→        label: "Tanggung Jawab Pihak Ketiga (TPL)",
   313→        rate: 0.01,
   314→        premium: tplPremium,
   315→        coverageAmount: tplCoverageAmount,
   316→      });
   317→      continue;
   318→    }
   319→
   320→    if (addonKey === "paDriver") {
   321→      const paDriverAmount = input.paDriverAmount || 10_000_000;
   322→      const paDriverPremium = Math.round(paDriverAmount * 0.005);
   323→      addonPremiums.push({
   324→        key: "paDriver",
   325→        label: "Perlindungan Jiwa Supir",
   326→        rate: 0.005,
   327→        premium: paDriverPremium,
   328→        coverageAmount: paDriverAmount,
   329→      });
   330→      continue;
   331→    }
   332→
   333→    if (addonKey === "paPassenger") {
   334→      const paPassengerPremium = 40000;
   335→      const paPassengerCount = input.paPassengerCount || 4;
   336→      addonPremiums.push({
   337→        key: "paPassenger",
   338→        label: `Perlindungan Jiwa Penumpang (${paPassengerCount} orang)`,
   339→        rate: 0,
   340→        premium: paPassengerPremium,
   341→        coverageAmount: paPassengerPremium,
   342→      });
   343→      continue;
   344→    }
   345→
   346→    const staticAddon = STATIC_ADDON_RATES[addonKey];
   347→    if (staticAddon) {
   348→      const applies = staticAddon.appliesTo.includes(input.coverageType) || staticAddon.appliesTo.includes("All");
   349→      if (applies) {
   350→        addonPremiums.push({
   351→          key: addonKey,
   352→          label: staticAddon.label,
   353→          rate: staticAddon.rate,
   354→          premium: Math.round(vehicleValue * staticAddon.rate),
   355→          coverageAmount: vehicleValue,
   356→        });
   357→      }
   358→    }
   359→  }
   360→
   361→  // 9. Totals
   362→  const totalAddonPremium = addonPremiums.reduce((sum, a) => sum + a.premium, 0);
   363→  const totalPremiumBeforeDiscount = basePremium + totalAddonPremium;
   364→  const discountPercent = 25; // Default discount
   365→  const discountAmount = Math.round(totalPremiumBeforeDiscount * (discountPercent / 100));
   366→  const adminFee = 50000;
   367→  const policyFee = 0;
   368→  const totalPremium = totalPremiumBeforeDiscount - discountAmount + adminFee + policyFee;
   369→
   370→  // 10. OTR range
   371→  const OTR_RANGE_PERCENT = 0.15;
   372→  const otrMin = Math.round(vehicleValue * (1 - OTR_RANGE_PERCENT));
   373→  const otrMax = Math.round(vehicleValue * (1 + OTR_RANGE_PERCENT));
   374→  const formatRupiah = (amt: number) =>
   375→    new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", minimumFractionDigits: 0, maximumFractionDigits: 0 }).format(amt);
   376→
   377→  // 11. Eligibility
   378→  let isEligible = true;
   379→  let ineligibilityReason: string | undefined;
   380→  if (input.coverageType === "Comprehensive" && vehicleAge > 12) {
   381→    isEligible = false;
   382→    ineligibilityReason = `Kendaraan berusia ${vehicleAge} tahun. All Risk/Comprehensive maksimal 12 tahun.`;
   383→  } else if (input.coverageType === "TLO" && vehicleAge > 15) {
   384→    isEligible = false;
   385→    ineligibilityReason = `Kendaraan berusia ${vehicleAge} tahun. TLO maksimal 15 tahun.`;
   386→  }
   387→
   388→  // 12. Partner calculation
   389→  const defaultModifiers: Record<string, number> = {
   390→    partnerSinarmasModifier: 1.0,
   391→    partnerMultiArthaGlobalModifier: 1.0,
   392→    partnerACAModifier: 1.0,
   393→    partnerMegaInsuranceModifier: 1.0,
   394→    partnerZurichsyariahModifier: 1.0,
   395→    partnerTuguModifier: 1.0,
   396→    partnerSahabatModifier: 1.0,
   397→    partnerOonaModifier: 1.0,
   398→  };
   399→
   400→  const partners = PARTNERS.map((partner) => {
   401→    const modifier = defaultModifiers[partner.key] ?? 1.0;
   402→    // Filter bengkelAuthorized if vehicle age exceeds partner's max
   403→    const bengkelExcluded = partner.bengkelResmiMaxYears != null && vehicleAge > partner.bengkelResmiMaxYears;
   404→    const filteredAddonPremiums = bengkelExcluded
   405→      ? addonPremiums.filter(a => a.key !== "bengkelAuthorized")
   406→      : addonPremiums;
   407→    const addonTotal = filteredAddonPremiums.reduce((sum, a) => sum + a.premium, 0);
   408→    const adjustedBasePremium = Math.round(basePremium * modifier);
   409→    const adjustedTotalBeforeDiscount = adjustedBasePremium + addonTotal;
   410→    const adjustedDiscountAmount = Math.round(adjustedTotalBeforeDiscount * (discountPercent / 100));
   411→    const estimatedPremium = adjustedTotalBeforeDiscount - adjustedDiscountAmount + adminFee + policyFee;
   412→    return {
   413→      name: partner.name,
   414→      modifier,
   415→      addonModifier: 1.0,
   416→      adminFee,
   417→      bengkelAuthorizedExcluded: bengkelExcluded,
   418→      estimatedPremium,
   419→      benefits: partner.benefits,
   420→      facilities: partner.facilities,
   421→      availableAddOns: partner.availableAddOns,
   422→      breakdown: {
   423→        basePremium: adjustedBasePremium,
   424→        addOnPremium: addonTotal,
   425→        addons: filteredAddonPremiums,
   426→        totalPremiumBeforeDiscount: adjustedTotalBeforeDiscount,
   427→        discountPercent,
   428→        discountAmount: adjustedDiscountAmount,
   429→        adminFee,
   430→        policyFee,
   431→      },
   432→    };
   433→  });
   434→
   435→  return {
   436→    success: true as const,
   437→    dataAvailable: true,
   438→    source: input.vehicleValueOverride ? "manual_override_static" : "pricelist_static_fallback",
   439→
   440→    vehicleValue,
   441→    vehicleAge,
   442→    vehicleTypeCategory,
   443→    wilayah,
   444→    wilayahName: wilayah === 1 ? "Wilayah 1 (Sumatera — Rate Tertinggi)" : wilayah === 2 ? "Wilayah 2 (Jabodetabek & Bandung)" : "Wilayah 3 (Wilayah Lainnya)",
   445→    plateCity,
   446→
   447→    isEligible,
   448→    ineligibilityReason,
   449→
   450→    otrRange: {
   451→      min: otrMin,
   452→      max: otrMax,
   453→      display: `${formatRupiah(otrMin)} – ${formatRupiah(otrMax)}`,
   454→    },
   455→
   456→    coverageType: input.coverageType,
   457→    baseRate,
   458→    loadingRate,
   459→    effectiveRate,
   460→    basePremium,
   461→
   462→    addOns: addonPremiums,
   463→    addOnPremium: totalAddonPremium,
   464→
   465→    totalPremiumBeforeDiscount,
   466→    discountPercent,
   467→    discountAmount,
   468→    adminFee,
   469→    policyFee,
   470→    totalPremium,
   471→
   472→    partners,
   473→  };
   474→}

   476→// POST /api/vehicles/premium — Real Insurance Quotation Engine
   477→// Vehicle values always from static JSON; rates/partners from DB with static fallback
   478→export async function POST(request: NextRequest) {
   479→  const start = Date.now();
   480→  try {
   481→    const body = await request.json();
   482→    const {
   483→      brand,
   484→      modelDescription,
   485→      vehicleYear,
   486→      coverageType,
   487→      addOns,
   488→      vehicleValueOverride,
   489→      plateCode,
   490→      vehicleTypeCategory,
   491→      tplCoverageAmount,
   492→      paDriverAmount,
   493→      paPassengerCount,
   494→      paPassengerAmount,
   495→    } = body as {
   496→      brand: string;
   497→      modelDescription: string;
   498→      vehicleYear: number;
   499→      coverageType: "TLO" | "AllRisk" | "Comprehensive";
   500→      addOns: string[];
   501→      vehicleValueOverride?: number;
   502→      plateCode?: string;
   503→      vehicleTypeCategory?: string;
   504→      tplCoverageAmount?: number;
   505→      paDriverAmount?: number;
   506→      paPassengerCount?: number;
   507→      paPassengerAmount?: number;
   508→    };
   509→
   510→    // Validate required fields
   511→    if (!brand || !modelDescription || !vehicleYear || !coverageType) {
   512→      return NextResponse.json(
   513→        { error: "Missing required fields: brand, modelDescription, vehicleYear, coverageType" },
   514→        { status: 400 }
   515→      );
   516→    }
   517→
   518→    // Normalize coverage type
   519→    const normalizedCoverage: "Comprehensive" | "TLO" =
   520→      coverageType === "TLO" ? "TLO" : "Comprehensive";
   521→
   522→    // ─── Try DB-powered calculation first (vehicle values from JSON, rates from DB) ───
   523→    try {
   524→      const input: QuotationInput = {
   525→        coverageType: normalizedCoverage,
   526→        vehicleBrand: brand,
   527→        vehicleModel: modelDescription,
   528→        vehicleYear: Number(vehicleYear),
   529→        plateCode: plateCode || "B",
   530→        vehicleValue: vehicleValueOverride || undefined,
   531→        addOns: Array.isArray(addOns) ? addOns : [],
   532→        vehicleTypeCategory: vehicleTypeCategory as QuotationInput["vehicleTypeCategory"],
   533→        tplCoverageAmount: tplCoverageAmount || 10_000_000,
   534→        paDriverAmount: paDriverAmount || 10_000_000,
   535→        paPassengerCount: paPassengerCount || 4,
   536→        paPassengerAmount: paPassengerAmount || 10_000_000,
   537→      };
   538→
   539→      const result = await calculatePremium(input);
   540→
   541→      const calcElapsed = Date.now() - start;
   542→      console.log(`[PERF] POST /api/vehicles/premium calculatePremium: ${calcElapsed}ms (${brand} ${modelDescription} ${vehicleYear} ${normalizedCoverage})`);
   543→
   544→      if (!result.success) {
   545→        return NextResponse.json(
   546→          {
   547→            error: result.error,
   548→            dataAvailable: false,
   549→            isEligible: result.isEligible,
   550→            ineligibilityReason: result.ineligibilityReason,
   551→            ...(result.vehicleValue > 0 ? {
   552→              vehicleValue: result.vehicleValue,
   553→              otrRange: result.otrRange,
   554→            } : {}),
   555→          },
   556→          { status: result.vehicleValue > 0 ? 200 : 404 }
   557→        );
   558→      }
   559→
   560→      // ─── Partner Comparison Calculation ───
   561→      // Try to fetch active partners from the database first
   562→      const dbPartners = await getActivePartnersFromDB();
   563→
   564→      // If DB partners are available, use them; otherwise fall back to hardcoded + rateSettings
   565→      let partners: Array<{
   566→        name: string;
   567→        modifier: number;
   568→        estimatedPremium: number;
   569→        benefits: string[];
   570→        facilities: string[];
   571→        availableAddOns: string[];
   572→      }>;
   573→
   574→      if (dbPartners && dbPartners.length > 0) {
   575→        // Use database-sourced partners (respects active/inactive status)
   576→        partners = dbPartners.map((partner) => {
   577→          // Calculate addon total with per-partner rate overrides
   578→          const overrideMap = new Map(partner.addonRateOverrides?.map(o => [o.addonKey, o]) ?? []);
   579→          let addonTotal = 0;
   580→
   581→          // Filter out bengkelAuthorized if vehicle age exceeds partner's max
   582→          const vehicleAge = result.vehicleAge ?? 0;
   583→          const bengkelExcluded = partner.bengkelResmiMaxYears != null && vehicleAge > partner.bengkelResmiMaxYears;
   584→
   585→          for (const addon of result.addons) {
   586→            // Skip bengkelAuthorized if vehicle is too old for this partner
   587→            if (addon.key === "bengkelAuthorized" && bengkelExcluded) continue;
   588→
   589→            const override = overrideMap.get(addon.key);
   590→            if (override && override.rate > 0) {
   591→              // Partner-specific rate override — recalculate premium using override rate
   592→              addonTotal += Math.round(result.vehicleValue * override.rate);
   593→            } else {
   594→              // No override — apply addonModifier to the global addon premium
   595→              addonTotal += Math.round(addon.premium * (partner.addonModifier ?? 1.0));
   596→            }
   597→          }
   598→          const adjustedBasePremium = Math.round(result.basePremium * partner.modifier);
   599→          const adjustedTotalBeforeDiscount = adjustedBasePremium + addonTotal;
   600→          const adjustedDiscountAmount = Math.round(adjustedTotalBeforeDiscount * (result.discountPercent / 100));
   601→          const partnerAdminFee = partner.adminFee ?? result.adminFee;
   602→          const estimatedPremium = adjustedTotalBeforeDiscount - adjustedDiscountAmount + partnerAdminFee + result.policyFee;
   603→
   604→          // Build per-partner addon breakdown (with overrides applied, filter bengkel if excluded)
   605→          const partnerAddons = result.addons
   606→            .filter((addon) => !(addon.key === "bengkelAuthorized" && bengkelExcluded))
   607→            .map((addon) => {
   608→            const override = overrideMap.get(addon.key);
   609→            if (override && override.rate > 0) {
   610→              return { ...addon, premium: Math.round(result.vehicleValue * override.rate) };
   611→            }
   612→            return { ...addon, premium: Math.round(addon.premium * (partner.addonModifier ?? 1.0)) };
   613→          });
   614→
   615→          return {
   616→            name: partner.name,
   617→            modifier: partner.modifier,
   618→            addonModifier: partner.addonModifier ?? 1.0,
   619→            adminFee: partnerAdminFee,
   620→            bengkelAuthorizedExcluded: bengkelExcluded,
   621→            estimatedPremium,
   622→            benefits: partner.benefits,
   623→            facilities: partner.facilities,
   624→            availableAddOns: partner.availableAddOns,
   625→            breakdown: {
   626→              basePremium: adjustedBasePremium,
   627→              addOnPremium: addonTotal,
   628→              addons: partnerAddons,
   629→              totalPremiumBeforeDiscount: adjustedTotalBeforeDiscount,
   630→              discountPercent: result.discountPercent,
   631→              discountAmount: adjustedDiscountAmount,
   632→              adminFee: partnerAdminFee,
   633→              policyFee: result.policyFee,
   634→            },
   635→          };
   636→        });
   637→      } else {
   638→        // Fallback: use hardcoded PARTNERS with rateSettings modifiers
   639→        let partnerModifiers: Array<{ key: string; value: number }> = [];
   640→        try {
   641→          partnerModifiers = await db.rateSettings.findMany({
   642→            where: {
   643→              category: "partner_modifier",
   644→              isActive: true,
   645→            },
   646→          });
   647→        } catch {
   648→          // DB partner modifiers not available — use defaults
   649→        }
   650→
   651→        const modifierMap = new Map<string, number>();
   652→        for (const setting of partnerModifiers) {
   653→          modifierMap.set(setting.key, setting.value);
   654→        }
   655→
   656→        const defaultModifiers: Record<string, number> = {
   657→          partnerSinarmasModifier: 1.0,
   658→          partnerMultiArthaGlobalModifier: 1.0,
   659→          partnerACAModifier: 1.0,
   660→          partnerMegaInsuranceModifier: 1.0,
   661→          partnerZurichsyariahModifier: 1.0,
   662→          partnerTuguModifier: 1.0,
   663→          partnerSahabatModifier: 1.0,
   664→          partnerOonaModifier: 1.0,
   665→        };
   666→
   667→        partners = PARTNERS.map((partner) => {
   668→          const modifier = modifierMap.get(partner.key) ?? defaultModifiers[partner.key] ?? 1.0;
   669→          // Filter bengkelAuthorized if vehicle age exceeds partner's max
   670→          const vehicleAge = result.vehicleAge ?? 0;
   671→          const bengkelExcluded = partner.bengkelResmiMaxYears != null && vehicleAge > partner.bengkelResmiMaxYears;
   672→          const filteredAddons = bengkelExcluded
   673→            ? result.addons.filter(a => a.key !== "bengkelAuthorized")
   674→            : result.addons;
   675→          const addonTotal = filteredAddons.reduce((sum, a) => sum + a.premium, 0);
   676→          const adjustedBasePremium = Math.round(result.basePremium * modifier);
   677→          const adjustedTotalBeforeDiscount = adjustedBasePremium + addonTotal;
   678→          const adjustedDiscountAmount = Math.round(adjustedTotalBeforeDiscount * (result.discountPercent / 100));
   679→          const estimatedPremium = adjustedTotalBeforeDiscount - adjustedDiscountAmount + result.adminFee + result.policyFee;
   680→          return {
   681→            name: partner.name,
   682→            modifier,
   683→            addonModifier: 1.0,
   684→            adminFee: result.adminFee,
   685→            bengkelAuthorizedExcluded: bengkelExcluded,
   686→            estimatedPremium,
   687→            benefits: partner.benefits,
   688→            facilities: partner.facilities,
   689→            availableAddOns: partner.availableAddOns,
   690→            breakdown: {
   691→              basePremium: adjustedBasePremium,
   692→              addOnPremium: addonTotal,
   693→              addons: filteredAddons,
   694→              totalPremiumBeforeDiscount: adjustedTotalBeforeDiscount,
   695→              discountPercent: result.discountPercent,
   696→              discountAmount: adjustedDiscountAmount,
   697→              adminFee: result.adminFee,
   698→              policyFee: result.policyFee,
   699→            },
   700→          };
   701→        });
   702→      }
   703→
   704→      return NextResponse.json({
   705→        dataAvailable: true,
   706→        source: input.vehicleValue ? "manual_override" : "static_json_db_rates",
   707→
   708→        vehicleValue: result.vehicleValue,
   709→        vehicleAge: result.vehicleAge,
   710→        vehicleTypeCategory: result.vehicleTypeCategory,
   711→        wilayah: result.wilayah,
   712→        wilayahName: result.wilayah === 1 ? "Wilayah 1 (Sumatera — Rate Tertinggi)" : result.wilayah === 2 ? "Wilayah 2 (Jabodetabek & Bandung)" : "Wilayah 3 (Wilayah Lainnya)",
   713→        plateCity: result.plateCity,
   714→
   715→        isEligible: result.isEligible,
   716→        ineligibilityReason: result.ineligibilityReason,
   717→
   718→        otrRange: result.otrRange,
   719→
   720→        coverageType: normalizedCoverage,
   721→        baseRate: result.baseRate,
   722→        loadingRate: result.loadingRate,
   723→        effectiveRate: result.effectiveRate,
   724→        basePremium: result.basePremium,
   725→
   726→        addOns: result.addons,
   727→        addOnPremium: result.addons.reduce((sum, a) => sum + a.premium, 0),
   728→
   729→        totalPremiumBeforeDiscount: result.totalPremiumBeforeDiscount,
   730→        discountPercent: result.discountPercent,
   731→        discountAmount: result.discountAmount,
   732→        adminFee: result.adminFee,
   733→        policyFee: result.policyFee,
   734→        totalPremium: result.totalPremium,
   735→
   736→        partners,
   737→      });
   738→    } catch (dbError) {
   739→      console.error("DB calculation failed, falling back to static:", dbError instanceof Error ? dbError.message : String(dbError));
   740→      // Fall through to static calculation below
   741→    }
   742→
   743→    // ─── Static Fallback Calculation (when DB is unavailable) ───
   744→    console.log("Using static fallback for premium calculation");
   745→    const staticResult = calculateStaticPremium({
   746→      brand,
   747→      modelDescription,
   748→      vehicleYear: Number(vehicleYear),
   749→      coverageType: normalizedCoverage,
   750→      plateCode: plateCode || "B",
   751→      addOns: Array.isArray(addOns) ? addOns : [],
   752→      vehicleValueOverride,
   753→      vehicleTypeCategory: vehicleTypeCategory || undefined,
   754→      tplCoverageAmount,
   755→      paDriverAmount,
   756→      paPassengerCount,

   759→    if (!staticResult.success) {
   760→      return NextResponse.json(
   761→        { error: staticResult.error, dataAvailable: false },
   762→        { status: 404 }
   763→      );
   764→    }
   765→
   766→    return NextResponse.json(staticResult);
   767→
   768→  } catch (error) {
   769→    const errorDetail = error instanceof Error ? error.message : String(error);
   770→    console.error("POST /api/vehicles/premium error:", errorDetail);
   771→    return NextResponse.json(
   772→      { error: "Gagal menghitung premi. Silakan coba lagi atau hubungi kami.", detail: process.env.NODE_ENV === "development" ? errorDetail : undefined },
   773→      { status: 500 }
   774→    );
   775→  }
